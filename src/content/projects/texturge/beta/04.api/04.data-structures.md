# 数据结构参考

Texturge Beta 中所有 USTRUCT 和非 UObject 数据结构的完整定义。

## USTRUCT（Blueprint 可见）

以下结构体标记 `USTRUCT(BlueprintType)`，可在 Blueprint 中使用。

### FBakedStage

**文件**: `Public/Animation/BakedAnimation.h`

单个动画阶段的烘焙结果。

```cpp
USTRUCT(BlueprintType)
struct FBakedStage
{
    UPROPERTY(BlueprintReadOnly)
    EAnimationStage StageType;        // Intro / Default / Outro

    UPROPERTY(BlueprintReadOnly)
    int32 GlyphCount;                 // 字形数量

    UPROPERTY(BlueprintReadOnly)
    float TotalDuration;              // 阶段总时长（秒）

    UPROPERTY(BlueprintReadOnly)
    TArray<FBakedGlyph> Glyphs;       // 字形数组
};
```

方法:
```cpp
bool IsValid() const;
int32 GetStartedGlyphCount(float StageTime) const;
```

### FBakedGlyph

**文件**: `Public/Animation/BakedAnimation.h`

单个字形的运行时烘焙数据。

```cpp
USTRUCT(BlueprintType)
struct FBakedGlyph
{
    UPROPERTY(BlueprintReadOnly)
    int32 OriginalIndex;              // 在源文本中的位置

    UPROPERTY(BlueprintReadOnly)
    FString Character;                // 字符本身

    UPROPERTY(BlueprintReadOnly)
    float StartTime;                  // 动画开始时间

    UPROPERTY(BlueprintReadOnly)
    float Duration;                   // 动画持续时间

    UPROPERTY(BlueprintReadOnly)
    bool bIsStatic;                   // 静态字形（无动画）

    UPROPERTY(BlueprintReadOnly)
    FGlyphAnimationState DefaultState; // 默认渲染状态

    UPROPERTY(BlueprintReadOnly)
    TArray<FPerLayerCurves> LayerCurves; // 多层曲线数据
};
```

方法:
```cpp
bool IsCurvesEmpty() const;
FVector2D GetCurvesTimeRange() const;
void CopyCurvesFrom(const FBakedGlyph& Other);
void ApplyCurvesScaleOffset(float Scale, float Offset);
void ApplyCurvesSeedJitter(float Jitter);
void OffsetCurvesBy(float Offset);
FBakedGlyph BuildCurveSet() const;
void AddLayer(const FPerLayerCurves& Layer);
bool HasStarted(float StageTime) const;
bool HasFinished(float StageTime) const;
void Evaluate(float StageTime, FGlyphAnimationState& OutState) const;
```

### FPerLayerCurves

**文件**: `Public/Animation/BakedAnimation.h`

单层 21 通道独立曲线存储。每个通道存储为 `TArray<FRichCurveKey>`。

```cpp
USTRUCT(BlueprintType)
struct FPerLayerCurves
{
    // 层配置
    UPROPERTY(BlueprintReadOnly) ETextAnimationBlendMode BlendMode;
    UPROPERTY(BlueprintReadOnly) EStagePlayDirection PlayDirection;
    UPROPERTY(BlueprintReadOnly) int32 NumberOfLoops;
    UPROPERTY(BlueprintReadOnly) float LayerDuration;

    // Float 通道 (3)
    UPROPERTY() TArray<FRichCurveKey> OpacityKeys;
    UPROPERTY() TArray<FRichCurveKey> LetterSpacingKeys;
    UPROPERTY() TArray<FRichCurveKey> RotationKeys;

    // Vector2D X/Y 通道 (10)
    UPROPERTY() TArray<FRichCurveKey> TranslationXKeys;
    UPROPERTY() TArray<FRichCurveKey> TranslationYKeys;
    UPROPERTY() TArray<FRichCurveKey> ScaleXKeys;
    UPROPERTY() TArray<FRichCurveKey> ScaleYKeys;
    UPROPERTY() TArray<FRichCurveKey> ShearXKeys;
    UPROPERTY() TArray<FRichCurveKey> ShearYKeys;
    UPROPERTY() TArray<FRichCurveKey> PivotXKeys;
    UPROPERTY() TArray<FRichCurveKey> PivotYKeys;
    UPROPERTY() TArray<FRichCurveKey> ShadowOffsetXKeys;
    UPROPERTY() TArray<FRichCurveKey> ShadowOffsetYKeys;

    // Color RGBA 通道 (8)
    UPROPERTY() TArray<FRichCurveKey> ColorRKeys;
    UPROPERTY() TArray<FRichCurveKey> ColorGKeys;
    UPROPERTY() TArray<FRichCurveKey> ColorBKeys;
    UPROPERTY() TArray<FRichCurveKey> ColorAKeys;
    UPROPERTY() TArray<FRichCurveKey> ShadowColorRKeys;
    UPROPERTY() TArray<FRichCurveKey> ShadowColorGKeys;
    UPROPERTY() TArray<FRichCurveKey> ShadowColorBKeys;
    UPROPERTY() TArray<FRichCurveKey> ShadowColorAKeys;
};
```

方法:
```cpp
bool IsEmpty() const;
float GetEffectiveDuration() const;
void Evaluate(float LocalTime, FGlyphAnimationState& InOutState) const;
```

### FAnimationEntry

**文件**: `Public/Animation/AnimationEntry.h`

标签到蓝图的映射。参数覆盖现使用 6 种独立的类型参数映射，与 `UAnimParameterOverrides` 模板系统兼容。

```cpp
USTRUCT(BlueprintType)
struct FAnimationEntry
{
    UPROPERTY(BlueprintReadWrite)
    FName TagName;                    // 标签名称

    UPROPERTY(BlueprintReadWrite)
    UTextAnimationBlueprint* Type;    // 动画蓝图引用

    // 六种独立类型参数覆盖 Map
    UPROPERTY(BlueprintReadWrite)
    TMap<FName, float> ParameterOverrides;           // float 覆盖
    UPROPERTY(BlueprintReadWrite)
    TMap<FName, int32> IntParameterOverrides;        // int32 覆盖
    UPROPERTY(BlueprintReadWrite)
    TMap<FName, bool> BoolParameterOverrides;        // bool 覆盖
    UPROPERTY(BlueprintReadWrite)
    TMap<FName, FVector> VectorParameterOverrides;   // FVector 覆盖
    UPROPERTY(BlueprintReadWrite)
    TMap<FName, FLinearColor> ColorParameterOverrides; // FLinearColor 覆盖
    UPROPERTY(BlueprintReadWrite)
    TMap<FName, FVector2D> Vector2DParameterOverrides;  // FVector2D 覆盖
};
```

```cpp
struct FBakedAnimation
{
    FString SourceText;
    int32 GlyphCount;
    uint64 TextFingerprint;
    uint64 BlueprintFingerprint;

    TUniquePtr<FBakedStage> IntroStage;
    TUniquePtr<FBakedStage> DefaultStage;
    TUniquePtr<FBakedStage> OutroStage;

    FBakedStage* GetStage(EAnimationStage Type);
    const FBakedStage* GetStage(EAnimationStage Type) const;
    bool HasStage(EAnimationStage Type) const;
    FBakedStage* GetFirstValidStage();
    EAnimationStage GetNextStageAfter(EAnimationStage Current) const;
    float GetTotalDuration() const;
};
```

### FGlyphCurveSet

**文件**: `Public/Animation/GlyphCurveSet.h`

21 FRichCurve 容器，C++ 内部专用。不使用 UPROPERTY。

```cpp
struct FGlyphCurveSet
{
    // Float curves
    FRichCurve Opacity;
    FRichCurve LetterSpacing;
    FRichCurve Rotation;

    // Vector2D component curves
    FRichCurve TranslationX, TranslationY;
    FRichCurve ScaleX, ScaleY;
    FRichCurve ShearX, ShearY;
    FRichCurve PivotX, PivotY;
    FRichCurve ShadowOffsetX, ShadowOffsetY;

    // Color component curves
    FRichCurve ColorR, ColorG, ColorB, ColorA;
    FRichCurve ShadowColorR, ShadowColorG, ShadowColorB, ShadowColorA;

    void Evaluate(float Time, FGlyphAnimationState& OutState) const;
    void OffsetBy(float Offset);
    void CopyFrom(const FGlyphCurveSet& Other);
    void ApplyScaleOffset(float Scale, float Offset);
    void ApplySeedJitter(float Jitter);
    FVector2D GetTimeRange() const;
    bool IsEmpty() const;
};
```
